```bash
cat > defaults/main.yml <<'EOF'
---
action: install   # install | remove

wordpress_db: wordpress
wordpress_db_user: admin
wordpress_db_password: networknuts

wordpress_packages:
  - php
  - php-mysql
  - mariadb-server
  - apache2
  - python3-pymysql
  - unzip
  - wget

EOF

cat > tasks/main.yml <<'EOF'
---
# ALWAYS ensure MySQL Python driver exists
- name: ensure mysql python driver is installed
  ansible.builtin.apt:
    name: python3-pymysql
    state: present

- name: update apt cache
  ansible.builtin.apt:
    update_cache: yes
  when: action == "install"

- name: install wordpress dependencies
  ansible.builtin.apt:
    name: "{{ wordpress_packages }}"
    state: present
  when: action == "install"

- name: allow firewall ports
  community.general.ufw:
    rule: allow
    port: "{{ item }}"
    proto: tcp
  loop:
    - 80
    - 443
  when: action == "install"

- name: remove firewall rules
  community.general.ufw:
    rule: deny
    port: "{{ item }}"
    proto: tcp
  loop:
    - 80
    - 443
  when: action == "remove"

- name: create wordpress database
  community.mysql.mysql_db:
    name: "{{ wordpress_db }}"
    state: present
    login_unix_socket: /var/run/mysqld/mysqld.sock
  when: action == "install"

- name: remove wordpress database
  community.mysql.mysql_db:
    name: "{{ wordpress_db }}"
    state: absent
    login_unix_socket: /var/run/mysqld/mysqld.sock
  when: action == "remove"

- name: create wordpress database user
  community.mysql.mysql_user:
    name: "{{ wordpress_db_user }}"
    password: "{{ wordpress_db_password }}"
    priv: "{{ wordpress_db }}.*:ALL"
    state: present
    login_unix_socket: /var/run/mysqld/mysqld.sock
  when: action == "install"

- name: remove wordpress database user
  community.mysql.mysql_user:
    name: "{{ wordpress_db_user }}"
    state: absent
    login_unix_socket: /var/run/mysqld/mysqld.sock
  when: action == "remove"

- name: download wordpress
  ansible.builtin.get_url:
    url: https://wordpress.org/latest.zip
    dest: /tmp/wordpress.zip
  when: action == "install"

- name: extract wordpress
  ansible.builtin.unarchive:
    src: /tmp/wordpress.zip
    dest: /var/www/html/
    remote_src: yes
  when: action == "install"

- name: remove wordpress files
  ansible.builtin.file:
    path: /var/www/html/wordpress
    state: absent
  when: action == "remove"

- name: restart apache
  ansible.builtin.service:
    name: apache2
    state: restarted
    enabled: true
  when: action == "install"

- name: stop and disable apache
  ansible.builtin.service:
    name: apache2
    state: stopped
    enabled: false
  when: action == "remove"

# ⚠️ REMOVE PACKAGES ONLY AFTER DB CLEANUP
- name: remove wordpress dependencies
  ansible.builtin.apt:
    name: "{{ wordpress_packages }}"
    state: absent
    purge: yes
  when: action == "remove"
EOF

cat > include_role.yaml <<'EOF'
---
- name: workdpress role
  hosts: all
  become: true
  roles:
  - role-name
EOF

ansible-playbook include_role.yaml -i inventory.yaml
ansible-playbook include_role.yaml -i inventory.yaml -e action=remove
```
